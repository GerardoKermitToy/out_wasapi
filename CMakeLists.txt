cmake_minimum_required(VERSION 3.21)
project(out_wasapi VERSION 1.2.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Use Visual Studio C++ and a Windows SDK on Windows.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Classic Winamp needs an x86 DLL. Configure with -A Win32.")
endif()
enable_language(RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Windows SDK only; no vcpkg, MFC, ATL, Wasabi or Winamp SDK.
# Release uses MultiThreadedDLL (/MD); Debug uses MultiThreadedDebugDLL (/MDd).
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_compile_options(/W4 /permissive- /utf-8 /EHsc /Zi)
add_library(wasapi_core STATIC src/pcm.cpp)
target_include_directories(wasapi_core PUBLIC include src)
add_library(out_wasapi SHARED src/plugin.cpp src/engine.cpp src/audio_stream.cpp src/audio_guids.cpp src/settings.cpp
  src/dialog.cpp resources/plugin.rc resources/exports.def)
target_include_directories(out_wasapi PRIVATE include src resources)
target_link_libraries(out_wasapi PRIVATE wasapi_core ole32 uuid avrt comctl32 shell32 user32 gdi32)
target_link_options(out_wasapi PRIVATE /DEBUG:FULL /INCREMENTAL:NO /DYNAMICBASE /NXCOMPAT)
set_target_properties(out_wasapi PROPERTIES PREFIX "" OUTPUT_NAME "out_wasapi")
install(TARGETS out_wasapi RUNTIME DESTINATION Plugins)
install(FILES $<TARGET_PDB_FILE:out_wasapi> DESTINATION Symbols OPTIONAL)
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE wasapi_core)
  add_test(NAME core COMMAND core_tests)
  add_executable(plugin_host tests/plugin_host.cpp)
  target_include_directories(plugin_host PRIVATE include src resources)
  target_link_libraries(plugin_host PRIVATE user32 gdi32 gdiplus ole32)
  add_executable(mixer_probe tests/mixer_probe.cpp)
  target_link_libraries(mixer_probe PRIVATE ole32)
  add_executable(notification_tests tests/notification_tests.cpp src/audio_guids.cpp)
  target_include_directories(notification_tests PRIVATE src)
  target_link_libraries(notification_tests PRIVATE ole32 uuid)
  add_test(NAME notifications COMMAND notification_tests)
  add_executable(recovery_tests tests/recovery_tests.cpp src/engine.cpp src/audio_stream.cpp src/audio_guids.cpp src/settings.cpp)
  target_include_directories(recovery_tests PRIVATE include src)
  target_compile_definitions(recovery_tests PRIVATE OUT_WASAPI_TESTING)
  target_link_libraries(recovery_tests PRIVATE wasapi_core ole32 uuid avrt shell32 user32)
  add_dependencies(plugin_host out_wasapi)
  add_test(NAME abi COMMAND plugin_host --abi $<TARGET_FILE:out_wasapi>)
  # Audio integration intentionally opt-in: requires a real render endpoint.
endif()
